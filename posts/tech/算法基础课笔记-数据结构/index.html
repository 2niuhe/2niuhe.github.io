栈#

828. 模拟栈 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
)

var stk = make([]int, 100010)

// head是栈顶指针, head为0表示栈为空
var head = 0

func main() {
	in := bufio.NewReader(os.Stdin)
	var m int

	fmt.Scanf("%d", &m)

	var op string
	var val int
	for ; m > 0; m-- {
		fmt.Fscan(in, &op)

		switch op {
		case "push":
			fmt.Fscan(in, &val)
			// 移动栈顶指针后赋值
			head++
			stk[head] = val
		case "pop":
			head--
		case "empty":
			if head > 0 {
				// 队列非空
				fmt.Println("NO")
			} else {
				fmt.Println("YES")
			}
		default:
			fmt.Println(stk[head])
		}
	}
}

队列#

829. 模拟队列 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
)

var q = make([]int, 100010)
var head, tail = 0, -1

// 当head和tail相等，表示队列有一个元素
func main() {
	in := bufio.NewReader(os.Stdin)
	var m int

	fmt.Scanf("%d", &m)

	var op string
	var val int
	for ; m > 0; m-- {
		fmt.Fscan(in, &op)

		switch op {
		case "push":
			fmt.Fscan(in, &val)
			tail++
			q[tail] = val

		case "pop":
			head++
		case "empty":
			if head <= tail {
				// 队列非空
				fmt.Println("NO")
			} else {
				fmt.Println("YES")
			}
		default:
			fmt.Println(q[head])
		}
	}
}

单调栈#

830. 单调栈 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

// head是栈顶指针, head为0表示栈为空
var head = 0

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	// remove prefix and suffix , \n \t etc
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func writeArrInt(data []int, delim string) {
	fmt.Println(strings.Trim(strings.Join(strings.Split(fmt.Sprint(data), " "), delim), "[]"))
}

func main() {
	in := bufio.NewReader(os.Stdin)
	var m int

	fmt.Scanf("%d", &m)
	// 栈顶从1计数，最大容量m
	stk := make([]int, m+1)
	nums := make([]int, m)
	output := make([]int, m)

	readArrInt(in, nums)
	for idx, val := range nums {

		for head > 0 && stk[head] >= val {
			head--
		}

		if head == 0 {
			// 栈为空
			output[idx] = -1
		} else {
			output[idx] = stk[head]
		}

		head++
		stk[head] = val
	}

	writeArrInt(output, " ")
}

KMP算法#

831. KMP字符串 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
)

func main() {
	in := bufio.NewReader(os.Stdin)
	var n, m int
	var p, s []byte

	fmt.Fscan(in, &n, &p, &m, &s)
	// next idx位置存储该位置的最长公共前缀和后缀
	next := make([]int, n)

	for i, j := 1, 0; i < n; i++ {
		for j > 0 && p[i] != p[j] {
			j = next[j-1]
		}
		if p[i] == p[j] {
			j++
		}
		next[i] = j
	}

	for i, j := 0, 0; i < m; i++ {
		for j > 0 && s[i] != p[j] {
			j = next[j-1]
		}

		if s[i] == p[j] {
			j++
		}
		if j == n {
			fmt.Printf("%d ", i-n+1)
			j = next[j-1]
		}
	}
}

Trie树#

用来快速存储字符串集合

835. Trie字符串统计 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

const N = 100010

// 根节点不储存任何数据, 是空节点，idx=0
var cnt = make([]int, N)
var idx int
var son [][]int = make([][]int, N)

func insert(s string) {
	p := 0 // p表示节点游标

	// iter byte by byte
	for i := 0; i < len(s); i++ {
		u := s[i] - 'a'
		// 当一个节点没有字母，就创建
		if son[p][u] == 0 {
			// idx给每个节点唯一多编号，先自增
			idx++
			son[p][u] = idx
		}
		p = son[p][u]
	}
	// cnt记录某个节点结尾的字符串的个数
	cnt[p]++
}

func query(s string) int {
	p := 0
	for i := 0; i < len(s); i++ {
		u := s[i] - 'a'
		if son[p][u] == 0 {
			return 0
		}
		p = son[p][u]
	}

	return cnt[p]
}

func main() {

	for i := 0; i < N; i++ {
		son[i] = make([]int, 26)
	}

	reader := bufio.NewReader(os.Stdin)
	var n int
	fmt.Fscan(reader, &n)

	res := make([]string, 0, n)

	for ; n != 0; n-- {
		var op string
		var x string
		fmt.Fscan(reader, &op, &x)
		if op == "I" {
			insert(x)
		} else if op == "Q" {
			res = append(res, strconv.Itoa(query(x)))
		}
	}

	fmt.Printf("%s", strings.Join(res, "\n"))
}

并查集#

836. 合并集合 - AcWing题库

package main

import (
	"fmt"
)

const N = 100010

var p = make([]int, N)

func find(x int) int {
	if p[x] != x {
		p[x] = find(p[x])
	}
	return p[x]
}

func main() {
	var n, m int

	fmt.Scanf("%d%d", &n, &m)

	for i := 1; i <= n; i++ {
		p[i] = i
	}
	var op string
	var num1, num2 int

	for i := 0; i < m; i++ {
		fmt.Scanf("%s%d%d", &op, &num1, &num2)

		switch op {
		case "M":
			p[find(num1)] = find(num2)
		case "Q":
			if find(num1) == find(num2) {
				fmt.Println("Yes")
			} else {
				fmt.Println("No")
			}
		}
	}
}
