快速排序#

快排基于分治思想

  • 确定分界点x，取中间元素或随机选择
  • 调整区间，使得区间一侧小于等于x，另一侧大于等于x
  • 递归处理左右两段区间

785. 快速排序 - AcWing题库

package main

import "fmt"

func quick_sort(nums []int, l int, r int) {
	if l >= r {
		return
	}

	// pivot选择中点元素
	i, j, x := l-1, r+1, nums[(l+r)>>1]

	for i < j {
		for ok := true; ok; ok = nums[i] < x {
			i++
		}

		for ok := true; ok; ok = nums[j] > x {
			j--
		}

		if i < j {
			nums[i], nums[j] = nums[j], nums[i]
		}
	}

	quick_sort(nums, l, j)
	quick_sort(nums, j+1, r)
}

func main() {
	var n int
	fmt.Scanf("%d", &n)
	nums := make([]int, n, n)

	for i := 0; i < n; i++ {
		fmt.Scanf("%d", &nums[i])
	}

	quick_sort(nums, 0, n-1)

	for _, v := range nums {
		fmt.Printf("%d ", v)
	}
}

归并排序#

787. 归并排序 - AcWing题库

package main

import "fmt"

var nums, tmp []int

func merge_sort(nums []int, l int, r int) {
	if l >= r {
		return
	}

	mid := (l + r) >> 1

	merge_sort(nums, l, mid)
	merge_sort(nums, mid+1, r)

	k, i, j := 0, l, mid+1

	for i <= mid && j <= r {
		if nums[i] <= nums[j] {
			tmp[k] = nums[i]
			k++
			i++
		} else {
			tmp[k] = nums[j]
			k++
			j++
		}
	}

	for i <= mid {
		tmp[k] = nums[i]
		k++
		i++
	}

	for j <= r {
		tmp[k] = nums[j]
		k++
		j++
	}

	for i, j = l, 0; i <= r; {
		nums[i] = tmp[j]
		i++
		j++
	}
}

func main() {
	var n int
	fmt.Scanf("%d", &n)

	nums = make([]int, n, n)
	tmp = make([]int, n, n)

	for i := 0; i < n; i++ {
		fmt.Scanf("%d", &nums[i])
	}

	merge_sort(nums, 0, n-1)

	for i := 0; i < n; i++ {
		fmt.Printf("%d ", nums[i])
	}
}

前缀和#

795. 前缀和 - AcWing题库

package main

import "fmt"

var nums, sums []int

func main() {
	var n, m int
	fmt.Scanf("%d%d", &n, &m)

	nums = make([]int, n+1)
	sums = make([]int, n+1)

	for i := 1; i <= n; i++ {
		fmt.Scanf("%d", &nums[i])
	}

	// s[0] = 0 s[1] = a[0]
	for i := 1; i <= n; i++ {
		// 前缀和初始化
		sums[i] = sums[i-1] + nums[i]
	}

	l, r := 0, 0
	for i := 0; i < m; i++ {
		fmt.Scanf("%d%d", &l, &r)
		// 打印区间和
		fmt.Printf("%d\n", sums[r]-sums[l-1])
	}
}

差分#

797. 差分 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

var nums, b []int

func insert(l int, r int, c int) {
	b[l] += c
	b[r+1] -= c
}

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func main() {
	var n, m int
	fmt.Scanf("%d%d", &n, &m)

	nums = make([]int, n+2)
	b = make([]int, n+2)

	in := bufio.NewReader(os.Stdin)
	readArrInt(in, nums[1:])

	// 差分
	for i := 1; i <= n; i++ {
		insert(i, i, nums[i])
	}

	lrc := make([]int, 3)

	for i := 0; i < m; i++ {
		readArrInt(in, lrc)
		insert(lrc[0], lrc[1], lrc[2])
	}

	//原地求前缀和
	for i := 1; i <= n; i++ {
		b[i] += b[i-1]
	}

	for i := 1; i <= n; i++ {
		fmt.Printf("%d ", b[i])
	}

}

双指针算法#

799. 最长连续不重复子序列 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func max(a, b int) int {
	if a > b {
		return a
	}
	return b
}

func main() {
	var n int
	fmt.Scanf("%d", &n)

	nums := make([]int, n)
	// Init counter with n items
	// counter := make(map[int]int, n)

	// slice is faster
	counter := make([]int, 1e5+1)

	in := bufio.NewReader(os.Stdin)
	readArrInt(in, nums)

	res := 0

	//双指针 使用counter计数
	for i, j := 0, 0; i < n; i++ {
		counter[nums[i]]++
		for j < i && counter[nums[i]] > 1 {
			counter[nums[j]]--
			j++
		}
		res = max(res, i-j+1)
	}

	fmt.Println(res)
}

800. 数组元素的目标和 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func main() {
	var n, m, x int
	fmt.Scanf("%d%d%d", &n, &m, &x)

	numsA := make([]int, n)
	numsB := make([]int, m)

	in := bufio.NewReader(os.Stdin)
	readArrInt(in, numsA)
	readArrInt(in, numsB)

	//双指针
	for i, j := 0, m-1; i < n; i++ {
		for j >= 0 && numsA[i]+numsB[j] > x {
			j--
		}

		if j >= 0 && numsA[i]+numsB[j] == x {
			fmt.Printf("%d %d", i, j)
			// 唯一解
			break
		}
	}
}

2816. 判断子序列 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func main() {
	var n, m int
	fmt.Scanf("%d%d%d", &n, &m)

	if n > m {
		fmt.Println("No")
		return
	}

	numsA := make([]int, n)
	numsB := make([]int, m)

	in := bufio.NewReader(os.Stdin)
	readArrInt(in, numsA)
	readArrInt(in, numsB)

	//双指针
	i, j := 0, 0
	for i < n && j < m {
		if numsA[i] == numsB[j] {
			i++
		}
		j++
	}

	if i == n {
		fmt.Println("Yes")
	} else {
		fmt.Println("No")
	}
}

位运算#

801. 二进制中1的个数 - AcWing题库

package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func readArrInt(in *bufio.Reader, out []int) {
	line, _ := in.ReadString('\n')
	line = strings.TrimSpace(line)

	foo := strings.Split(line, " ")
	for i, bar := range foo {
		out[i], _ = strconv.Atoi(bar)
	}
}

func writeArrInt(data []int, delim string) {
	fmt.Println(strings.Trim(strings.Join(strings.Split(fmt.Sprint(data), " "), delim), "[]"))
}

func main() {
	var n int
	fmt.Scanf("%d", &n)

	nums := make([]int, n)
	counter := make([]int, n)

	in := bufio.NewReader(os.Stdin)
	readArrInt(in, nums)

	for idx, num := range nums {
		res := 0

		for num > 0 {
			num -= num & -num
			res++
		}
		counter[idx] = res
	}

	writeArrInt(counter, " ")
}
